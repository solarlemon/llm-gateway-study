cmake_minimum_required(VERSION 3.16)
project(llm_gateway LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_DIR "$ENV{HOME}/cpp_code/llama.cpp" CACHE PATH "llama.cpp directory")
find_package(Threads REQUIRED)
add_executable(gateway main.cpp "${LLAMA_DIR}/vendor/cpp-httplib/httplib.cpp")
target_include_directories(gateway PRIVATE "${LLAMA_DIR}/vendor/cpp-httplib" "${LLAMA_DIR}/vendor/nlohmann")
target_compile_options(gateway PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(gateway PRIVATE Threads::Threads)
